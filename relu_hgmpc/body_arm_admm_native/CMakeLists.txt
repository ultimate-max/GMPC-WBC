cmake_minimum_required(VERSION 3.22)
project(body_arm_admm_native LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(Eigen3 REQUIRED NO_MODULE)
find_package(Python3 REQUIRED COMPONENTS Interpreter Development.Module)
find_package(OpenMP REQUIRED)
set(PYBIND_HEADERS /home/ganhe/miniconda3/envs/frictionest/lib/python3.11/site-packages/torch/include CACHE PATH "pybind11 headers")
Python3_add_library(body_arm_admm_native MODULE solver.cpp WITH_SOABI)
target_include_directories(body_arm_admm_native PRIVATE ${PYBIND_HEADERS})
target_link_libraries(body_arm_admm_native PRIVATE Eigen3::Eigen OpenMP::OpenMP_CXX)
target_compile_options(body_arm_admm_native PRIVATE -O3)
target_compile_definitions(body_arm_admm_native PRIVATE EIGEN_DONT_PARALLELIZE EIGEN_NO_DEBUG)
option(BODY_ARM_ADMM_BLAS "Use BLAS with one thread per local block" ON)
option(BODY_ARM_ADMM_NATIVE_ARCH "Enable host CPU vector instructions" ON)
if(BODY_ARM_ADMM_BLAS)
  find_library(ADMM_BLAS NAMES blas openblas REQUIRED)
  target_link_libraries(body_arm_admm_native PRIVATE ${ADMM_BLAS})
  target_compile_definitions(body_arm_admm_native PRIVATE EIGEN_USE_BLAS ADMM_BLAS_ENABLED)
endif()
if(BODY_ARM_ADMM_NATIVE_ARCH)
  target_compile_options(body_arm_admm_native PRIVATE -march=native)
  target_compile_definitions(body_arm_admm_native PRIVATE ADMM_NATIVE_ARCH)
endif()
