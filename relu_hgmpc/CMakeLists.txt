cmake_minimum_required(VERSION 3.22)
project(relu_hgmpc LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(pinocchio 3 REQUIRED)
find_package(Python3 REQUIRED COMPONENTS Interpreter Development.Module)
set(ROOT ${CMAKE_CURRENT_SOURCE_DIR}/..)
execute_process(COMMAND ${Python3_EXECUTABLE} -c "import mujoco,pathlib; print(pathlib.Path(mujoco.__file__).parent)" OUTPUT_VARIABLE MJ OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
set(TORCH_HEADERS /home/ganhe/miniconda3/envs/frictionest/lib/python3.11/site-packages/torch/include CACHE PATH "pybind11 headers bundled with PyTorch")
file(GLOB MJLIB ${MJ}/libmujoco.so.*)
Python3_add_library(native MODULE native.cpp wbc_native.cpp ${ROOT}/src/robot.cpp WITH_SOABI)
target_include_directories(native PRIVATE ${ROOT}/src ${ROOT}/paper_repro ${MJ}/include ${TORCH_HEADERS})
target_link_libraries(native PRIVATE pinocchio::pinocchio ${MJLIB} ${CMAKE_DL_LIBS})
target_compile_options(native PRIVATE -O2)
set_target_properties(native PROPERTIES BUILD_RPATH "${MJ}")
