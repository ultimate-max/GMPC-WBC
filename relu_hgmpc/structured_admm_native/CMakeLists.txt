cmake_minimum_required(VERSION 3.22)
project(structured_admm_native LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(Eigen3 REQUIRED NO_MODULE)
find_package(Python3 REQUIRED COMPONENTS Interpreter Development.Module)
set(PYBIND_HEADERS /home/ganhe/miniconda3/envs/frictionest/lib/python3.11/site-packages/torch/include CACHE PATH "pybind11 headers")
Python3_add_library(structured_admm_native MODULE solver.cpp WITH_SOABI)
target_include_directories(structured_admm_native PRIVATE ${PYBIND_HEADERS})
target_link_libraries(structured_admm_native PRIVATE Eigen3::Eigen)
target_compile_options(structured_admm_native PRIVATE -O3)
target_compile_definitions(structured_admm_native PRIVATE EIGEN_DONT_PARALLELIZE EIGEN_NO_DEBUG)

option(STRUCTURED_ADMM_BLAS "Use system BLAS for dense products" ON)
option(STRUCTURED_ADMM_NATIVE_ARCH "Enable host CPU vector instructions" ON)
if(STRUCTURED_ADMM_BLAS)
  find_library(ADMM_BLAS NAMES blas openblas REQUIRED)
  target_link_libraries(structured_admm_native PRIVATE ${ADMM_BLAS})
  target_compile_definitions(structured_admm_native PRIVATE EIGEN_USE_BLAS ADMM_BLAS_ENABLED)
endif()
if(STRUCTURED_ADMM_NATIVE_ARCH)
  target_compile_options(structured_admm_native PRIVATE -march=native)
  target_compile_definitions(structured_admm_native PRIVATE ADMM_NATIVE_ARCH)
endif()
