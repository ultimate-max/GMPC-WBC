cmake_minimum_required(VERSION 3.22)
project(b2z1_geometric_mpc LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(pinocchio 3.0 REQUIRED)
set(OSQP_ENABLE_INTERRUPT OFF CACHE BOOL "No process-global signal handler in threaded solves" FORCE)
set(OSQP_USE_LONG ON CACHE BOOL "" FORCE)
set(OSQP_BUILD_DEMO_EXE OFF CACHE BOOL "" FORCE)
set(OSQP_ENABLE_PRINTING OFF CACHE BOOL "" FORCE)
set(OSQP_ENABLE_DERIVATIVES OFF CACHE BOOL "" FORCE)
set(OSQP_CODEGEN OFF CACHE BOOL "" FORCE)
add_subdirectory(third_party/osqp EXCLUDE_FROM_ALL)
find_package(OpenMP REQUIRED)
add_library(b2z1_control STATIC src/robot.cpp src/qp.cpp src/mpc.cpp src/subsystem.cpp src/warm_start.cpp)
target_include_directories(b2z1_control PUBLIC src)
target_link_libraries(b2z1_control PUBLIC pinocchio::pinocchio osqpstatic OpenMP::OpenMP_CXX)
target_compile_options(b2z1_control PRIVATE -Wall -Wextra)
add_executable(b2z1_demo src/main.cpp)
target_link_libraries(b2z1_demo PRIVATE b2z1_control)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
execute_process(COMMAND ${Python3_EXECUTABLE} -c "import mujoco,pathlib; print(pathlib.Path(mujoco.__file__).parent)"
  OUTPUT_VARIABLE MUJOCO_ROOT OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
file(GLOB MUJOCO_LIB "${MUJOCO_ROOT}/libmujoco.so.*")
add_executable(b2z1_closed_loop src/closed_loop.cpp)
target_include_directories(b2z1_closed_loop PRIVATE "${MUJOCO_ROOT}/include")
target_link_libraries(b2z1_closed_loop PRIVATE b2z1_control ${MUJOCO_LIB})
set_target_properties(b2z1_closed_loop PROPERTIES BUILD_RPATH "${MUJOCO_ROOT}")
enable_testing()
add_test(NAME model_and_solver COMMAND b2z1_demo ${CMAKE_SOURCE_DIR} --verify)
add_executable(b2z1_test_online src/test_online.cpp)
target_link_libraries(b2z1_test_online PRIVATE b2z1_control)
add_test(NAME online_derivatives_and_budgets COMMAND b2z1_test_online ${CMAKE_SOURCE_DIR})
add_test(NAME mujoco_central_closed_loop COMMAND b2z1_closed_loop ${CMAKE_SOURCE_DIR} --mode central --duration 2 --name test_central)
add_test(NAME mujoco_admm_closed_loop COMMAND b2z1_closed_loop ${CMAKE_SOURCE_DIR} --mode admm --duration 2 --outer 10 --name test_admm)
set_tests_properties(model_and_solver online_derivatives_and_budgets mujoco_central_closed_loop mujoco_admm_closed_loop
  PROPERTIES ENVIRONMENT "OPENBLAS_NUM_THREADS=1;OMP_DYNAMIC=FALSE" RUN_SERIAL TRUE)
