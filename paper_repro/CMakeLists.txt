cmake_minimum_required(VERSION 3.22)
project(paper_reproduction LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(pinocchio 3 REQUIRED)
find_package(OpenMP REQUIRED)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
set(ROOT ${CMAKE_CURRENT_SOURCE_DIR}/..)
set(ACADOS ${ROOT}/third_party/acados)
execute_process(COMMAND ${Python3_EXECUTABLE} -c "import mujoco,pathlib; print(pathlib.Path(mujoco.__file__).parent)" OUTPUT_VARIABLE MJ OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
file(GLOB MJLIB ${MJ}/libmujoco.so.*)
add_executable(paper_rti runtime.cpp ${ROOT}/src/robot.cpp)
target_include_directories(paper_rti PRIVATE ${ROOT}/src ${MJ}/include generated ${ACADOS}/include ${ACADOS}/include/blasfeo/include ${ACADOS}/include/hpipm/include)
target_link_libraries(paper_rti PRIVATE pinocchio::pinocchio OpenMP::OpenMP_CXX ${MJLIB} ${ACADOS}/lib/libacados.so ${ACADOS}/lib/libhpipm.so ${ACADOS}/lib/libblasfeo.so)
foreach(name central front rear arm)
  target_sources(paper_rti PRIVATE generated/diag_${name}.c)
  target_link_libraries(paper_rti PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/generated/${name}/libacados_ocp_solver_rti_${name}.so)
endforeach()
set_target_properties(paper_rti PROPERTIES BUILD_RPATH "${MJ};${ACADOS}/lib")
target_compile_options(paper_rti PRIVATE -O3)
